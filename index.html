async function pesquisarFilmes() { const pesquisa = campoBusca.value.trim(); if (!pesquisa) { mensagemBusca.textContent = "Digite o nome de um filme."; return; } resultadosBusca.innerHTML = ""; mensagemBusca.textContent = "Procurando filmes..."; try { const resposta = await fetch( `https://api.themoviedb.org/3/search/movie?api_key=${TMDB_API_KEY}&language=pt-BR&query=${encodeURIComponent(pesquisa)}&include_adult=false` ); if (!resposta.ok) throw new Error("Erro na pesquisa."); const dados = await resposta.json(); if (!dados.results.length) { mensagemBusca.textContent = "Nenhum filme encontrado."; return; } mensagemBusca.textContent = "Carregando filmes e plataformas..."; for (const filme of dados.results.slice(0, 10)) { const generos = (filme.genre_ids || []) .map(id => generosTMDB[id]) .filter(Boolean) .join(", "); const poster = filme.poster_path ? `https://image.tmdb.org/t/p/w500${filme.poster_path}` : ""; // BUSCAR PLATAFORMAS NO BRASIL let plataformas = []; try { const respostaPlataformas = await fetch( `https://api.themoviedb.org/3/movie/${filme.id}/watch/providers?api_key=${TMDB_API_KEY}` ); if (respostaPlataformas.ok) { const dadosPlataformas = await respostaPlataformas.json(); const brasil = dadosPlataformas.results?.BR; if (brasil) { const provedores = [ ...(brasil.flatrate || []), ...(brasil.free || []), ...(brasil.ads || []), ...(brasil.rent || []), ...(brasil.buy || []) ]; plataformas = [ ...new Set( provedores.map(p => p.provider_name) ) ]; } } } catch (erro) { console.error("Erro ao buscar plataformas:", erro); } const textoPlataformas = plataformas.length ? plataformas.join(", ") : "Disponibilidade não informada no Brasil."; resultadosBusca.innerHTML += `
${poster ? `` : ""}

${escapar(filme.title)}

${escapar(filme.overview || "Sinopse indisponível.")}

Lançamento: ${escapar(filme.release_date || "Não informado")}

Gêneros: ${escapar(generos || "Não informado")}

Onde assistir: ${escapar(textoPlataformas)}

`; } mensagemBusca.textContent = "Escolha o filme que deseja adicionar:"; } catch (erro) { console.error(erro); mensagemBusca.textContent = "Erro ao pesquisar filmes. Confira a API."; } }
